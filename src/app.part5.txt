on vertices, explicit normals and smoothing groups\no WeldedMesh\n`;
  const bc=base.getAttribute('color');
  for(let i=0;i<bp.count;i++){
    if(bc)s+=`v ${bp.getX(i)} ${bp.getY(i)} ${bp.getZ(i)} ${bc.getX(i)} ${bc.getY(i)} ${bc.getZ(i)}\n`;
    else s+=`v ${bp.getX(i)} ${bp.getY(i)} ${bp.getZ(i)}\n`;
  }
  for(let i=0;i<dn.count;i++)s+=`vn ${dn.getX(i)} ${dn.getY(i)} ${dn.getZ(i)}\n`;
  let normalCursor=1;
  for(let gi=0;gi<state.groups.length;gi++){
    const gr=state.groups[gi];if(!gr.visible||(state.isolated>=0&&gi!==state.isolated))continue;
    s+=`\ng ${gr.name.replace(/\s+/g,'_')}\n`;
    s+=gr.mode==='flat'?'s off\n':`s ${gi+1}\n`;
    for(let t=0;t<state.triGroups.length;t++){
      if(state.triGroups[t]!==gi)continue;
      const a=bix[t*3]+1,b=bix[t*3+1]+1,c=bix[t*3+2]+1;
      s+=`f ${a}//${normalCursor} ${b}//${normalCursor+1} ${c}//${normalCursor+2}\n`;
      normalCursor+=3;
    }
  }
  download(new Blob([s],{type:'text/plain'}),stem()+'_welded_smoothed.obj');
  status(`OBJ exported • ${state.weldedBefore-state.weldedAfter} duplicate vertices welded`);
};
