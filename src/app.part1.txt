function ensureIndexed(g){
  if(g.index) return g.clone();
  const count=g.getAttribute('position').count;
  const arr=count>65535?new Uint32Array(count):new Uint16Array(count);
  for(let i=0;i<count;i++) arr[i]=i;
  const c=g.clone(); c.setIndex(new THREE.BufferAttribute(arr,1)); return c;
}

function normalizeMergeAttributes(geometries){
  const names=new Set();
  for(const g of geometries)for(const name of Object.keys(g.attributes))names.add(name);
  for(const g of geometries){
    const count=g.getAttribute('position').count;
    for(const name of names){
      if(g.getAttribute(name))continue;
      if(name==='color'){const a=new Float32Array(count*3);a.fill(1);g.setAttribute('color',new THREE.BufferAttribute(a,3));}
      else if(name==='uv')g.setAttribute('uv',new THREE.BufferAttribute(new Float32Array(count*2),2));
      else if(name==='normal')g.computeVertexNormals();
    }
  }
}

function weldGeometry(source,tolerance=1e-6){
  const g=ensureIndexed(source),position=g.getAttribute('position'),index=g.index.array;
  const inv=1/Math.max(tolerance,1e-12),lookup=new Map(),positions=[],remap=new Uint32Array(position.count);
  for(let i=0;i<position.count;i++){
    const x=position.getX(i),y=position.getY(i),z=position.getZ(i);
    const key=`${Math.round(x*inv)},${Math.round(y*inv)},${Math.round(z*inv)}`;
    let next=lookup.get(key);
    if(next===undefined){next=positions.length/3;lookup.set(key,next);positions.push(x,y,z);}
    remap[i]=next;
  }
  const out=new THREE.BufferGeometry();
  out.setAttribute('position',new THREE.Float32BufferAttribute(positions,3));
  const IndexType=positions.length/3>65535?Uint32Array:Uint16Array,newIndex=new IndexType(index.length);
  for(let i=0;i<index.length;i++)newIndex[i]=remap[index[i]];
  out.setIndex(new THREE.BufferAttribute(newIndex,1));
  const color=g.getAttribute('color');
  if(color){
    const sums=new Float32Array((positions.length/3)*3),counts=new Uint32Array(positions.length/3);
    for(let i=0;i<position.count;i++){const n=remap[i];sums[n*3]+=color.getX(i);sums[n*3+1]+=color.getY(i);sums[n*3+2]+=color.getZ(i);counts[n]++;}
    for(let i=0;i<counts.length;i++){const c=counts[i]||1;sums[i*3]/=c;sums[i*3+1]/=c;sums[i*3+2]/=c;}
    out.setAttribute('color',new THREE.BufferAttribute(sums,3));
  }
  return out;
}
